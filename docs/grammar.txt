PROGRAM ::= LANGUAGE_CONSTRUCT*
LANGUAGE_CONSTRUCT ::= FUNCTION | STRUCT | ANNOTATION_PROFILE | IMPORT | STATEMENT 

STATEMENT ::=
    | ASSIGNMENT
    | EXPRESSION

ASSIGNMENT ::= 'let' 'mut'? <identifier> TYPE ASSIGNMENT_SUFFIX ';'
TYPE = ':' TYPE_BASE
TYPE_BASE ::=
    | <identifier>
    | '*' TYPE_BASE 
    | '[' TYPE_BASE (';' <number>)? ']'

EXPRESSION ::=
    | BINARY_EXPR ';'
    | IF
    | WHILE
    | SCOPE

BINARY_EXPR ::= UNARY BINARY_EXPR_RHS
BINARY_EXPR_RHS ::= (BIN_OPERATOR UNARY)*
BIN_OPERATOR ::=
    | '==' | '!=' | '<' | '<=' | '>' | '>=' # Comparison
    | '&&' | '||'                           # Logical
    | '+' | '-' | '*' | '/' | '%'           # Arithmetic
    | '<<' | '>>' | '&' | '|' | '^'         # Bitwise

UNARY ::= UN_OPERATOR UNARY | PRIMARY_EXPRESSION
UN_OPERATOR ::= 
    | '+' | '-' | '!' # Arithmetic
    | '~'             # Bitwise
    | '*' | '&'       # Pointer

PRIMARY_EXPRESSION ::= PRIMARY_EXPRESSION_BASE ( CALL_SUFFIX | INDEX_SUFFIX | MEMBER_SUFFIX )* ASSIGNMENT_SUFFIX?
PRIMARY_EXPRESSION_BASE ::=
    | <number> | 'true' | 'false' | <char> | <string>
    | '[' (EXPRESSION (',' EXPRESSION)*)? ']' # Array Literal
    | <identifier>
    | '(' EXPRESSION ')'

CALL_SUFFIX   ::= '(' (EXPRESSION (',' EXPRESSION)*)? ')'
INDEX_SUFFIX  ::= '[' EXPRESSION ']'
MEMBER_SUFFIX ::= '.' <identifier>
ASSIGNMENT_SUFFIX = '=' (SCOPE | EXPRESSION)

ANNOTATION ::= ANNOTATION_OPTIZ | ANNOTATION_USE
ANNOTATION_OPTIZ ::= '@optiz' '(' ANNOTATION_REPEATED ')'
ANNOTATION_USE   ::= '@use' '(' <string>+ ')'
CONTRACT ::= '@contract' '(' ANNOTATION_REPEATED ')'

ANNOTATION_REPEATED ::= ANNOTATION_BASE (',' ANNOTATION_BASE)*
ANNOTATION_BASE  ::= <identifier> '=' ( <token> | '{' <token>+ '}' )  # <token> accepts all language tokens, except curly braces

SCOPE ::= ANNOTATION* CONTRACT? '{' STATEMENT* EXPRESSION? '}'

IF ::= 'if' EXPRESSION 'then' SCOPE ('else' (IF | SCOPE))?

WHILE ::= 'while' EXPRESSION 'do' SCOPE

FUNCTION ::= 'fn' <identifier> '(' PARAMS? ')' TYPE SCOPE
PARAMS ::= <identifier> TYPE (',' <identifier> TYPE)*

STRUCT ::= 'struct' <identifier> '{' (<identifier> TYPE)* '}'

ANNOTATION_PROFILE ::= '@profile' '(' name '=' <string> ',' ANNOTATION_REPEATED ')'

IMPORT ::= 'import' <string>
